# THE AIRPLANE TEST, IN THE AIR ONLY (owner 2026-11-28: "jump ahead right away to the part where the plane is flying in the
# air"): the same aircraft and cameras as airplane-test.txt from 10,000 ft on - the turn, then the landing. Nothing is
# recorded before the aircraft is up there (the take-off happens unseen, far from the camera).
#   bash Tools/airplane_test.sh video flight
cmd r.VSync 0
framegen on
# (steady pacing: 30 drawn frames, each followed by its in-between picture = exactly 60 evenly spaced on screen and in the
#  60 fps recording; uncapped the game draws 40-50 unevenly and the recording judders - owner 11-28 "remove the jitter")
cmd t.MaxFPS 30
speed 1 1
day 172
hour 20.8
weather 1
wait 20
hideui
flight takeoff a320
flightwait 150 HB-TOFF high
orbit 42 95 95 2 0 HB-TOFF
wait 2
sound record airplane-flight
# ===== 1 10,000 ft: tracking, round to ahead, the banked turn (wings dip), behind, level again
shot4k high-track
orbit 48 95 -15 3 6 HB-TOFF
flightwait 30 HB-TOFF turn
wait 4
shot4k turn-ahead-bank
orbit 46 -15 75 5 14 HB-TOFF
wait 7
shot4k turn-inside-wing-down
orbit 55 150 215 4 16 HB-TOFF
wait 8
shot4k turn-behind
orbit 48 -40 -20 1 8 HB-TOFF
flightwait 70 HB-TOFF high
wait 2
shot4k turn-level-again
# ===== 2 landing: approach from ahead and beside, low by the touchdown zone (main gear smoke, the nose let down gently), rollout
flight land a320 60
orbit 50 -35 -35 4 0 HB-LAND
wait 4
shot4k land-approach-ahead
orbit 42 -35 80 2 14 HB-LAND
wait 10
shot4k land-approach-side
wait 18
orbit 36 105 105 g1.6 0 HB-LAND
flightwait 45 HB-LAND rollout
shot4k land-touchdown-smoke
orbit 34 100 115 g1.6 6 HB-LAND
wait 3
shot4k land-nose-down
orbit 40 115 165 g1.8 10 HB-LAND
wait 5
shot4k land-rollout-reversers
wait 4
# ===== end
sound stop
wait 2
lives
parkoff
hour live
weather live
cmd t.MaxFPS 0
quit
